function coalVal(sMask,metric){
var total=0;for(var key in coalitionConv){var k=Number(key);if((k&sMask)===k&&k>0)total+=coalitionConv[k][metric];}return total;
}
var fact=[1];for(var i=1;i<=N;i++)fact[i]=fact[i-1]*i;
var metrics=['trx','nbrx','nrx'];
var shapley={};metrics.forEach(function(m){shapley[m]={};});
for(var ci=0;ci<N;ci++){
var chBit=1<<ci;
var others=[];for(var j=0;j<N;j++){if(j!==ci)others.push(j);}
var M=others.length;var totalSubsets=1<<M;
metrics.forEach(function(metric){
var phi=0;
for(var mask=0;mask<totalSubsets;mask++){
var cMask=0,sSize=0;
for(var b=0;b<M;b++){if(mask&(1<<b)){cMask|=(1<<others[b]);sSize++;}}
var w=fact[sSize]*fact[N-sSize-1]/fact[N];
phi+=w*(coalVal(cMask|chBit,metric)-coalVal(cMask,metric));
}
shapley[metric][channelList[ci]]=phi;
});
}
var ltData=aggregateForModel(filtered,'lastTouch');
var totalScripts={trx:0,nbrx:0,nrx:0};
ltData.forEach(function(r){totalScripts.trx+=r.trx;totalScripts.nbrx+=r.nbrx;totalScripts.nrx+=r.nrx;});
return channelOrder.map(function(ch){
var res={channel:ch};
metrics.forEach(function(metric){
var totalRaw=0;for(var k in shapley[metric])totalRaw+=shapley[metric][k];
var w=totalRaw>0?(shapley[metric][ch]||0)/totalRaw:0;
var total=totalScripts[metric];
res[metric]=Math.round(total*w);
});
return res;
});
}

function computeIncrKpis(){
var mk=currentMetric;
var ctrl=incrData.overall.control;
var exp=incrData.overall.exposed;
var monthKeys=getFilteredMonthKeys();
var ctrlScripts,expScripts,ctrlHcps=ctrl.hcps,expHcps=exp.hcps;
if(monthKeys&&incrData.monthly){
ctrlScripts=0;expScripts=0;
monthKeys.forEach(function(k){
var m=incrData.monthly[k];
if(m){
if(m.ctrl)ctrlScripts+=m.ctrl[mk]||0;
if(m.exp)expScripts+=m.exp[mk]||0;
}
});
}else{
ctrlScripts=ctrl[mk]||0;
expScripts=exp[mk]||0;
}
var ctrlRate=ctrlHcps>0?ctrlScripts/ctrlHcps:0;
var expRate=expHcps>0?expScripts/expHcps:0;
var lift=ctrlRate>0?((expRate-ctrlRate)/ctrlRate)*100:0;
var incrScripts=Math.max(0,expScripts-(expHcps*ctrlRate));
var totalSpend=incrData.overall.totalSpend||0;
var cpis=incrScripts>0?totalSpend/incrScripts:0;
return {ctrlRate:ctrlRate,expRate:expRate,lift:lift,incrScripts:Math.round(incrScripts),cpis:cpis,ctrlHcps:ctrlHcps,expHcps:expHcps,ctrlScripts:ctrlScripts,expScripts:expScripts};
}
function computeChannelIncr(){
var mk=currentMetric;
var rateKey=mk==='trx'?'ctrlTrxRate':mk==='nbrx'?'ctrlNbrxRate':'ctrlNrxRate';
var raw=(incrData.byChannel||[]).map(function(ch){
var ctrlRate=ch[rateKey]||0;
var expRate=ch.hcpCount>0?(ch[mk]||0)/ch.hcpCount:0;
var lift=ctrlRate>0?((expRate-ctrlRate)/ctrlRate)*100:0;
var incrScripts=Math.max(0,(ch[mk]||0)-(ch.hcpCount*ctrlRate));
var spend=(incrData.spendByChannel||{})[ch.channel]||0;
var cpis=incrScripts>0?spend/incrScripts:0;
return{channel:ch.channel,hcpCount:ch.hcpCount,expRate:expRate,ctrlRate:ctrlRate,lift:lift,rawIncr:Math.round(incrScripts),incrScripts:Math.round(incrScripts),cpis:Math.round(cpis),spend:spend};
});
var overallKpi=computeIncrKpis();
var overallIncr=overallKpi.incrScripts;
var rawSum=raw.reduce(function(s,c){return s+c.rawIncr;},0);
if(rawSum>0&&overallIncr>0){
raw.forEach(function(c){
c.incrScripts=Math.round(c.rawIncr*(overallIncr/rawSum));
});
}
return raw;
}
function getMtaIncrementalData(){
var metrics=['trx','nbrx','nrx'];
var result={};
metrics.forEach(function(mk){
var rateKey=mk==='trx'?'ctrlTrxRate':mk==='nbrx'?'ctrlNbrxRate':'ctrlNrxRate';
var raw=(incrData.byChannel||[]).map(function(ch){
var ctrlRate=ch[rateKey]||0;
var incrScripts=Math.max(0,(ch[mk]||0)-(ch.hcpCount*ctrlRate));
return{channel:ch.channel,rawIncr:Math.round(incrScripts)};
});
var ctrl=incrData.overall.control;
var exp=incrData.overall.exposed;
var ctrlHcps=ctrl.hcps||1;
var overallCtrlRate=ctrlHcps>0?(ctrl[mk]||0)/ctrlHcps:0;
var overallIncr=Math.max(0,Math.round((exp[mk]||0)-(exp.hcps*overallCtrlRate)));
var rawSum=raw.reduce(function(s,c){return s+c.rawIncr;},0);
var scale=(rawSum>0&&overallIncr>0)?overallIncr/rawSum:1;
raw.forEach(function(c){
if(!result[c.channel]) result[c.channel]={channel:c.channel};
result[c.channel][mk]=Math.round(c.rawIncr*scale);
});
});
return Object.values(result);
}
